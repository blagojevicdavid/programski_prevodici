using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
 
namespace SOV2.Models
{
    public class Igrac
    {
        public string Ime { get; set; }
        public string Prezime { get; set; }
        public string BrojDresa { get; set; }
        public Pozicija Pozicija { get; set; }
        public bool Reprezentativac { get; set; }
 
        public Igrac() { }
 
        public Igrac(string ime, string prezime, string brojDresa,
                     Pozicija pozicija, bool reprezentativac)
        {
            Ime = ime;
            Prezime = prezime;
            BrojDresa = brojDresa;
            Pozicija = pozicija;
            Reprezentativac = reprezentativac;
        }
    }
}

FAJL: Controllers/HomeController.cs
========================
using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Web.Mvc;
using SOV2.Models;
 
namespace SOV2.Controllers
{
    public class HomeController : Controller
    {
        // GET: Home - pocetna stranica sa formom za unos i pretragu
        public ActionResult Index()
        {
            return View();
        }
 
        // POST: Home/Dodaj - dodavanje novog igraca
        [HttpPost]
        public ActionResult Dodaj(Igrac igrac)
        {
            List<Igrac> igraci = (List<Igrac>)HttpContext.Application["igraci"];
 
            // Provera da li igrac sa tim brojem dresa vec postoji
            Igrac postojeci = igraci.Find(i => i.BrojDresa.Equals(igrac.BrojDresa));
            if (postojeci != null)
            {
                ViewBag.Greska = "Uneti broj dresa nije jedinstven!";
                return View("Index");
            }
 
            igraci.Add(igrac);
            HttpContext.Application["igraci"] = igraci;
 
            return RedirectToAction("Spisak");
        }
 
        // GET: Home/Spisak - tabelarni prikaz svih igraca
        public ActionResult Spisak()
        {
            List<Igrac> igraci = (List<Igrac>)HttpContext.Application["igraci"];
            return View(igraci);
        }
 
        // GET: Home/Pretrazi?prezime=... - pretraga po prezimenu (sadrzi)
        public ActionResult Pretrazi(string prezime)
        {
            List<Igrac> igraci = (List<Igrac>)HttpContext.Application["igraci"];
 
            if (string.IsNullOrEmpty(prezime))
            {
                return View("Spisak", igraci);
            }
 
            List<Igrac> pronadjeni = igraci.FindAll(i => i.Prezime.ToLower().Contains(prezime.ToLower()));
 
            return View("Spisak", pronadjeni);
        }
    }
}

FAJL: Global.asax.cs   (PREPRAVITI postojeci Application_Start)
======================
using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Web.Mvc;
using System.Web.Optimization;
using System.Web.Routing;
using SOV2.Models;
 
namespace SOV2
{
    public class MvcApplication : System.Web.HttpApplication
    {
        protected void Application_Start()
        {
            AreaRegistration.RegisterAllAreas();
 
            RouteConfig.RegisterRoutes(RouteTable.Routes);
 
 
            // Inicijalna lista igraca (test podaci sa Slike 2)
            List<Igrac> igraci = new List<Igrac>
            {
                new Igrac("Petar", "Petrovic", "5", Pozicija.Bek, false),
                new Igrac("Ana", "Anic", "12", Pozicija.Centar, false),
                new Igrac("Marko", "Markovic", "18", Pozicija.Krilo, true)
            };
 
            HttpContext.Current.Application["igraci"] = igraci;
        }
    }
}

NAPOMENE:
================================================================================
 
1. Default ruta (App_Start/RouteConfig.cs) - NE diraj, ostavi default.
   Default je { controller = "Home", action = "Index" } sto je tacno ono
   sto nam treba (otvara Index.cshtml na pocetnoj).
 
2. Layout - ne koristi se _Layout.cshtml jer projekat je kreiran kao Empty + MVC
   pa Shared folder ne postoji. Stranice su ciste bez layouta.
 
3. Test podaci (Petar Petrovic / Ana Anic / Marko Markovic) ucitavaju se
   automatski pri pokretanju aplikacije iz Global.asax.cs - tako da odmah
   kad odes na spisak vec ces videti 3 igraca.
 
4. Tok aplikacije:
   - Otvara se Index.cshtml (forma za unos + pretraga)
   - Klik na "Dodaj" -> POST na /Home/Dodaj
     * Ako broj dresa vec postoji -> ostaje na Index sa porukom "Uneti broj
       dresa nije jedinstven!" (Slika 3)
     * Ako broj dresa ne postoji -> RedirectToAction na Spisak (Slika 2)
   - Klik na "Trazi" -> GET na /Home/Pretrazi?prezime=X
     * Vraca View "Spisak" sa igracima kod kojih prezime SADRZI unetu vrednost
   - Klik na "Nazad" na spisku -> vraca na Index.cshtml
 
5. Validacija - tekst poruke "Uneti broj dresa nije jedinstven!" je tacno
   onako kako je trazeno na Slici 3.
 
6. Checkbox za Reprezentativac:
   - Kada je cekiran, salje vrednost "true" -> model binding automatski setuje
     bool na true.
   - Kada nije cekiran, checkbox ne salje nista -> bool ostaje default false.
   - U tabeli se prikazuje kao "Da" / "Ne" preko ternary operatora.
 
7. Pretraga po prezimenu radi po SUBSTRING-u (Contains), case-insensitive,
   sto znaci da "ovic" pronalazi i "Petrovic" i "Markovic".
 
8. Sva polja se salju kao stringovi/enumi/bool - automatic model binding ce
   da popuni Igrac objekat.
