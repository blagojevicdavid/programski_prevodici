#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct sat_st {
    char model[21];
    char brend[16];
    float cena;
    int vodootpornost_m;
    struct sat_st *sledeci;
} sat_st;

sat_st* dodaj_sortirano(sat_st *glava, sat_st *novi) {
    if (glava == NULL || novi->cena <= glava->cena) {
        novi->sledeci = glava;
        return novi;
    }
    sat_st *tek = glava;
    while (tek->sledeci != NULL && tek->sledeci->cena < novi->cena) {
        tek = tek->sledeci;
    }
    novi->sledeci = tek->sledeci;
    tek->sledeci = novi;
    return glava;
}

sat_st* ucitaj_listu(const char *ime_datoteke) {
    FILE *f = fopen(ime_datoteke, "r");
    if (f == NULL) return NULL;

    sat_st *glava = NULL;
    char model[21], brend[16];
    float cena;
    int vodootpornost;

    while (fscanf(f, "%20s %15s %f %d", model, brend, &cena, &vodootpornost) == 4) {
        sat_st *novi = (sat_st*)malloc(sizeof(sat_st));
        if (novi == NULL) {
            fclose(f);
            // oslobodi listu
            while (glava) { sat_st *t = glava->sledeci; free(glava); glava = t; }
            return (sat_st*)-1; // signal za grešku memorije
        }
        strcpy(novi->model, model);
        strcpy(novi->brend, brend);
        novi->cena = cena;
        novi->vodootpornost_m = vodootpornost;
        novi->sledeci = NULL;
        glava = dodaj_sortirano(glava, novi);
    }

    fclose(f);
    return glava;
}

void oslobodi_listu(sat_st *glava) {
    while (glava) {
        sat_st *t = glava->sledeci;
        free(glava);
        glava = t;
    }
}

int main(int argc, char *argv[]) {
    if (argc != 6) {
        exit(3);
    }

    const char *ulaz = argv[1];
    const char *izlaz = argv[2];
    const char *brend = argv[3];
    int broj_komada = atoi(argv[4]);
    float budzet = atof(argv[5]);

    sat_st *glava = ucitaj_listu(ulaz);
    if (glava == (sat_st*)-1) {
        exit(4);
    }
    if (glava == NULL) {
        exit(1);
    }

    FILE *out = fopen(izlaz, "w");
    if (out == NULL) {
        oslobodi_listu(glava);
        exit(2);
    }

    // Pronađi najbolji sat brenda u okviru budžeta
    sat_st *najbolji = NULL;
    float min_razlika = -1;
    sat_st *tek = glava;

    while (tek != NULL) {
        if (strcmp(tek->brend, brend) == 0) {
            float ukupno = tek->cena * broj_komada;
            if (ukupno <= budzet) {
                float razlika = budzet - ukupno;
                if (min_razlika < 0 || razlika < min_razlika) {
                    min_razlika = razlika;
                    najbolji = tek;
                }
            }
        }
        tek = tek->sledeci;
    }

    fprintf(out, "Najbolji sat brenda %s:\n", brend);
    if (najbolji == NULL) {
        fprintf(out, "Nema odgovarajuceg sata.\n");
    } else {
        fprintf(out, "%s %s %.1f %d\n", najbolji->model, najbolji->brend,
                najbolji->cena, najbolji->vodootpornost_m);
    }

    // Vodootpornost najboljeg (ili 0 ako nema)
    int vodo_granica = (najbolji != NULL) ? najbolji->vodootpornost_m : 0;

    fprintf(out, "\nSatovi brenda '%s' sa vodootpornoscu od najmanje %dm:\n", brend, vodo_granica);

    int nadjen = 0;
    tek = glava;
    while (tek != NULL) {
        if (strcmp(tek->brend, brend) == 0 && tek->vodootpornost_m >= vodo_granica) {
            fprintf(out, "%s %s %.1f %d\n", tek->model, tek->brend,
                    tek->cena, tek->vodootpornost_m);
            nadjen = 1;
        }
        tek = tek->sledeci;
    }

    if (!nadjen) {
        fprintf(out, "Nema\n");
    }

    fclose(out);
    oslobodi_listu(glava);
    return 0;
}
