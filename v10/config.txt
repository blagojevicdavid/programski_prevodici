NAPOMENA: nema direktnog uzora na vjezbama za semanticku vrijednost
zajednickog tokena smjera; koristen obrazac iz v03 zadatak 5
(zajednicki token) i v04 (token sa vrijednoscu).

direction_statement
  : _DIRECTION _ID
      {
        if(lookup_symbol($2, VAR|PAR) == NO_INDEX)
          err("'%s' undeclared", $2);
      }
    _KRAJ literal _POCETAK literal
      {
        if(get_type($4) != get_type($6))
          err("incompatible types in direction statement");
        if($1 == 1 && atoi(get_name($6)) >= atoi(get_name($4)))
          err("start literal must be less than end literal");
        if($1 == 0 && atoi(get_name($6)) <= atoi(get_name($4)))
          err("start literal must be greater than end literal");
      }
    _IZVRSI statement
  ;
