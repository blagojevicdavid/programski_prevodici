-- Zadatak 1: nepokretnosti u Petrovaradinu sortirane rastuće po kvadraturi
SELECT IDN, ULICA, UL_BR FROM Nepokretnost WHERE OPSTINA = 'Petrovaradin' ORDER BY KVD ASC;

-- Zadatak 2: nepokretnosti sa kvadraturom između 50 i 100 (uključujući granice)
SELECT IDN, KVD, ULICA, UL_BR FROM Nepokretnost WHERE KVD BETWEEN 50 AND 100;

-- Zadatak 3: dodavanje kolona DATPOCIZG i DATZAVIZG u tabelu Nepokretnost
ALTER TABLE Nepokretnost ADD(DATPOCIZG date, DATZAVIZG date);

-- Zadatak 4: lica sa saldom i kategorijom zaduženosti
-- Pretpostavka: saldo tačno -15000 spada u 'Umereno zaduzena' (saldo "do -15.000")
SELECT l.IDL, l.IMEL, l.PRZL, pk.SALDO, CASE WHEN pk.SALDO >= 0 THEN 'Nezaduzena' WHEN pk.SALDO < 0 AND pk.SALDO >= -15000 THEN 'Umereno zaduzena' ELSE 'Zaduzena' END AS zaduzenost FROM Lice l INNER JOIN PoreskaKartica pk ON l.IDL = pk.IDL;

-- Zadatak 5: poreske kartice pravnih lica sa dugovanjem i njihove nepokretnosti
SELECT pk.IDK, l.IMEL, l.PRZL, n.KVD, n.ULICA, vn.NAZIVVN FROM PoreskaKartica pk INNER JOIN Lice l ON pk.IDL = l.IDL INNER JOIN Nepokretnost n ON l.IDL = n.IDL INNER JOIN VrstaNep vn ON n.IDVN = vn.IDVN WHERE l.VRSTAL = 'PRAVNO' AND pk.DUG > 0;

-- Zadatak 6: lica koja imaju Stan, a nemaju Kucu
SELECT IDL, IMEL, PRZL FROM Lice WHERE IDL IN (SELECT n.IDL FROM Nepokretnost n INNER JOIN VrstaNep vn ON n.IDVN = vn.IDVN WHERE vn.NAZIVVN = 'Stan') AND IDL NOT IN (SELECT n.IDL FROM Nepokretnost n INNER JOIN VrstaNep vn ON n.IDVN = vn.IDVN WHERE vn.NAZIVVN = 'Kuca');

-- Zadatak 7: nepokretnosti sa kvadraturom manjom od prosečne za svoju vrstu
-- Pretpostavka: kolona BROJ iz postavke je u šemi UL_BR
SELECT n.IDN, n.KVD, n.ULICA, n.UL_BR, vn.IDVN, vn.NAZIVVN FROM Nepokretnost n INNER JOIN VrstaNep vn ON n.IDVN = vn.IDVN WHERE n.KVD < (SELECT AVG(n1.KVD) FROM Nepokretnost n1 WHERE n1.IDVN = n.IDVN);

-- Zadatak 8: povećanje kamate za 5% na karticama sa dugom
UPDATE PoreskaKartica SET KAMATA = KAMATA*1.05 WHERE DUG > 0;

-- Zadatak 9: broj nepokretnosti, broj vrsta, ukupna kvadratura i saldo po licu (analitičke funkcije), ukupna kvadratura < 200
-- Pretpostavka: svako lice ima jednu poresku karticu; lica bez nepokretnosti ulaze sa kvadraturom 0
WITH podaci_lica AS (SELECT DISTINCT l.IDL, l.IMEL, l.PRZL, COUNT(n.IDN) OVER (PARTITION BY l.IDL) AS br_nep, COUNT(DISTINCT n.IDVN) OVER (PARTITION BY l.IDL) AS br_vrsta, NVL(SUM(n.KVD) OVER (PARTITION BY l.IDL), 0) AS uk_kvd, pk.SALDO FROM Lice l LEFT OUTER JOIN Nepokretnost n ON l.IDL = n.IDL INNER JOIN PoreskaKartica pk ON l.IDL = pk.IDL) SELECT IMEL, PRZL, br_nep, br_vrsta, uk_kvd, SALDO FROM podaci_lica WHERE uk_kvd < 200;

-- Zadatak 10: pogled sa ukupnim dugom po grupama vlasnika, zatim prikaz grupa sa dugom većim od prosečnog
-- Napomena: postavka traži kreiranje pogleda i prikaz, pa su potrebne dve naredbe (CREATE VIEW + SELECT)
CREATE OR REPLACE VIEW dug_po_grupama (Grupa, UkDug) AS SELECT 'Fizicka lica - Kuca', NVL(SUM(pk.DUG), 0) FROM PoreskaKartica pk WHERE pk.IDL IN (SELECT n.IDL FROM Nepokretnost n INNER JOIN VrstaNep vn ON n.IDVN = vn.IDVN INNER JOIN Lice l ON n.IDL = l.IDL WHERE l.VRSTAL = 'FIZICKO' AND vn.NAZIVVN = 'Kuca') UNION SELECT 'Pravna lica - Poslovni prostor', NVL(SUM(pk.DUG), 0) FROM PoreskaKartica pk WHERE pk.IDL IN (SELECT n.IDL FROM Nepokretnost n INNER JOIN VrstaNep vn ON n.IDVN = vn.IDVN INNER JOIN Lice l ON n.IDL = l.IDL WHERE l.VRSTAL = 'PRAVNO' AND vn.NAZIVVN = 'Poslovni prostor') UNION SELECT 'Fizicka lica bez nepokretnosti', NVL(SUM(pk.DUG), 0) FROM PoreskaKartica pk INNER JOIN Lice l ON pk.IDL = l.IDL WHERE l.VRSTAL = 'FIZICKO' AND l.IDL NOT IN (SELECT IDL FROM Nepokretnost WHERE IDL IS NOT NULL);
SELECT * FROM dug_po_grupama WHERE UkDug > (SELECT AVG(DUG) FROM PoreskaKartica);
